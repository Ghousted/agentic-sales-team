# Lead-Scraping Worker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the real engine behind the `scrape` capability — a standalone Python worker driving a real browser (browser-use + Gemini) that finds ICP-matched prospects and writes them to the `leads` table, triggered on-demand from a capability-aware "Find leads" button.

**Architecture:** The Next.js app and a new local Python worker share **only the Neon database** (plus the filesystem for screenshot PNGs). The app enqueues `kind='scrape'` jobs and polls their status; the Python worker polls `jobs`, runs browser-use, and writes `leads` + `job_steps`. The existing Node runner (`/api/jobs/run`) is narrowed to claim only `outreach`, so the two runners never contend.

**Tech Stack:** Next.js 14.2.15 / TypeScript / Drizzle ORM / Neon Postgres (neon-http) / Gemini REST (existing); Python 3.11+ / browser-use / Playwright Chromium / Gemini / psycopg3 (new worker).

```ts
export async function enqueueJob(input: { agent: AgentSnapshot; kind: string; capabilities?: string[]; leadIds?: string[]; batchSize?: number; target?: number }): Promise<{ jobId: string } | { error: string }> {
  if (!isDbConfigured()) return { error: "Server not configured." };
  const userId = await uid();
  if (!userId) return { error: "Not signed in." };
  // Capability gating: an agent can only run a kind its capabilities include.
  if (input.capabilities && !input.capabilities.includes(input.kind)) {
    return { error: "This agent can't run " + input.kind + "." };
  }
  const [row] = await getDb().insert(jobs).values({
    userId, agentId: input.agent.id, kind: input.kind, status: "queued",
    params: { agent: input.agent, leadIds: input.leadIds ?? null, batchSize: input.batchSize ?? 5, target: input.target ?? 10 },
  }).returning({ id: jobs.id });
  return { jobId: row.id };
}
```

```python
"""Neon Postgres access for the scrape worker. Raw SQL against the exact
snake_case column names in agentic-salesforce/lib/db/schema.ts."""
import json
import os
from pathlib import Path

import psycopg
from dotenv import load_dotenv

# The app's .env.local lives one dir up (agentic-salesforce/.env.local).
load_dotenv(Path(__file__).resolve().parent.parent / ".env.local")

DATABASE_URL = os.environ.get("DATABASE_URL")


def connect() -> psycopg.Connection:
    if not DATABASE_URL:
        raise SystemExit("DATABASE_URL missing (expected in agentic-salesforce/.env.local)")
    # autocommit: each helper is its own unit of work, simplest for a poller.
    return psycopg.connect(DATABASE_URL, autocommit=True)


def claim_next_scrape_job(conn) -> dict | None:
    """Atomically claim the oldest queued scrape job. Returns None if none."""
    with conn.cursor() as cur:
        cur.execute(
            """
            UPDATE jobs SET status='running', started_at=now()
            WHERE id = (
              SELECT id FROM jobs
              WHERE status='queued' AND kind='scrape'
              ORDER BY created_at ASC
              LIMIT 1
              FOR UPDATE SKIP LOCKED
            )
            RETURNING id, user_id, agent_id, params
            """
        )
        row = cur.fetchone()
    if not row:
        return None
    return {"id": row[0], "user_id": row[1], "agent_id": row[2], "params": row[3]}


def get_scrape_settings(conn, user_id, agent_id) -> dict:
    with conn.cursor() as cur:
        cur.execute(
            "SELECT settings FROM agent_config WHERE user_id=%s AND agent_id=%s",
            (user_id, agent_id),
        )
        row = cur.fetchone()
    settings = (row[0] if row else None) or {}
    return settings.get("scrape", {}) if isinstance(settings, dict) else {}


def lead_exists(conn, user_id, agent_id, profile_url) -> bool:
    if not profile_url:
        return False
    with conn.cursor() as cur:
        cur.execute(
            "SELECT 1 FROM leads WHERE user_id=%s AND agent_id=%s AND profile_url=%s LIMIT 1",
            (user_id, agent_id, profile_url),
        )
        return cur.fetchone() is not None


def insert_lead(conn, user_id, agent_id, prospect: dict) -> None:
    with conn.cursor() as cur:
        cur.execute(
            """
            INSERT INTO leads (user_id, agent_id, name, title, company, email,
                               source, profile_url, platform)
            VALUES (%s, %s, %s, %s, %s, %s, 'scrape', %s, %s)
            """,
            (
                user_id, agent_id,
                prospect.get("name", "").strip() or "Unknown",
                prospect.get("title", "") or "",
                prospect.get("company", "") or "",
                prospect.get("email", "") or "",
                prospect.get("profileUrl", "") or "",
                prospect.get("platform", "") or "",
            ),
        )


def insert_job_step(conn, job_id, user_id, idx, caption, url, screenshot_path) -> None:
    with conn.cursor() as cur:
        cur.execute(
            """
            INSERT INTO job_steps (job_id, user_id, idx, caption, url, screenshot_path)
            VALUES (%s, %s, %s, %s, %s, %s)
            """,
            (job_id, user_id, idx, caption[:500], url[:1000], screenshot_path),
        )


def update_job_result(conn, job_id, user_id, result: dict) -> None:
    with conn.cursor() as cur:
        cur.execute(
            "UPDATE jobs SET result=%s WHERE id=%s AND user_id=%s",
            (json.dumps(result), job_id, user_id),
        )


def finish_job(conn, job_id, user_id, result: dict) -> None:
    with conn.cursor() as cur:
        cur.execute(
            "UPDATE jobs SET status='done', finished_at=now(), result=%s WHERE id=%s AND user_id=%s",
            (json.dumps(result), job_id, user_id),
        )


def fail_job(conn, job_id, user_id, error: str) -> None:
    with conn.cursor() as cur:
        cur.execute(
            "UPDATE jobs SET status='failed', finished_at=now(), error=%s WHERE id=%s AND user_id=%s",
            (error[:500], job_id, user_id),
        )


if __name__ == "__main__":
    # Self-test: connectivity + how many scrape jobs are queued.
    c = connect()
    with c.cursor() as cur:
        cur.execute("SELECT count(*) FROM jobs WHERE kind='scrape' AND status='queued'")
        print("DB OK — queued scrape jobs:", cur.fetchone()[0])
```

```python
"""Browser-use + Gemini scrape engine. Given an ICP task, drives a real browser
to find prospects and returns structured Prospect rows. Emits per-step frames
via an on_step callback. Verified against browser-use 0.13.3."""
import base64
import os
from pathlib import Path

from pydantic import BaseModel

from browser_use import Agent, ChatGoogle, BrowserProfile, BrowserSession

PROFILE_DIR = str(Path(__file__).resolve().parent / ".browser-profile")


class Prospect(BaseModel):
    name: str
    title: str = ""
    company: str = ""
    profileUrl: str = ""
    platform: str = ""
    email: str = ""  # found-only; blank unless visibly on the page


class Prospects(BaseModel):
    prospects: list[Prospect]


def build_task(settings: dict, agent: dict, target: int) -> str:
    industry = (settings.get("industry") or "").strip()
    titles = (settings.get("titles") or "").strip()
    geo = (settings.get("geo") or "").strip()
    icp_lines = []
    if industry: icp_lines.append(f"- Industry / market: {industry}")
    if titles: icp_lines.append(f"- Job titles / roles: {titles}")
    if geo: icp_lines.append(f"- Geography: {geo}")
    icp = "\n".join(icp_lines) or f"- Fit for this agent's goal: {agent.get('goal', '')}"
    return f"""You are a lead-research assistant. Find up to {target} real sales
prospects that match this ideal customer profile:
{icp}

Search across Reddit, the public web, and open directories. If you are logged in
to Facebook, you may also use Facebook groups/pages. Prefer sources that expose a
person's name, role, company, and a link to their profile or page.

For each prospect capture: name, title, company, a profileUrl (link to their
profile/page — REQUIRED), and the platform it came from (reddit / facebook / web).
Only include an email if it is VISIBLY shown on the page — never guess or
construct one; leave it blank otherwise.

Stop when you have {target} prospects or you have exhausted reasonable sources.
Return them as structured data."""


async def run_scrape(task: str, target: int, on_step) -> list[Prospect]:
    max_steps = min(60, 6 * target + 10)
    # Persistent, headed profile → reuses the user's logged-in sessions (Facebook, etc.).
    session = BrowserSession(
        browser_profile=BrowserProfile(user_data_dir=PROFILE_DIR, headless=False)
    )

    counter = {"i": 0}

    async def _on_step(browser_state_summary, agent_output, step_number):
        i = counter["i"]
        counter["i"] += 1
        caption = (getattr(agent_output, "next_goal", "")
                   or getattr(agent_output, "thinking", "") or "").strip()
        url = getattr(browser_state_summary, "url", "") or ""
        png = None
        b64 = getattr(browser_state_summary, "screenshot", None)
        if b64:
            try:
                png = base64.b64decode(b64)
            except Exception:
                png = None
        try:
            on_step(i, caption or f"Step {i + 1}", url, png)
        except Exception:
            pass

    agent = Agent(
        task=task,
        llm=ChatGoogle(model="gemini-2.5-flash",
                       api_key=os.environ.get("GEMINI_API_KEY"),
                       thinking_budget=0),
        browser_session=session,
        output_model_schema=Prospects,
        register_new_step_callback=_on_step,
    )
    history = await agent.run(max_steps=max_steps)
    result = history.structured_output  # property → Prospects | None
    if result:
        return result.prospects
    return []
```

```python
"""Scrape worker entrypoint: poll jobs, run browser-use, persist leads + steps."""
import asyncio
import time
from io import BytesIO
from pathlib import Path

from PIL import Image

import db
from scraper import build_task, run_scrape

# Screenshots are served by Next from agentic-salesforce/public/runs/<jobId>/.
PUBLIC_RUNS = Path(__file__).resolve().parent.parent / "public" / "runs"
POLL_SECONDS = 4


def _save_screenshot(job_id, idx, png_bytes) -> str:
    """Write a downscaled PNG, return the public path (/runs/<jobId>/step-<idx>.png)."""
    if not png_bytes:
        return ""
    out_dir = PUBLIC_RUNS / str(job_id)
    out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"step-{idx}.png"
    try:
        img = Image.open(BytesIO(png_bytes))
        if img.width > 1000:
            ratio = 1000 / img.width
            img = img.resize((1000, int(img.height * ratio)))
        img.save(path, "PNG")
    except Exception:
        path.write_bytes(png_bytes)  # fall back to raw bytes
    return f"/runs/{job_id}/step-{idx}.png"


def run_job(conn, job):
    job_id, user_id, agent_id = job["id"], job["user_id"], job["agent_id"]
    params = job["params"] or {}
    agent = params.get("agent", {})
    target = int(params.get("target", 10))
    settings = db.get_scrape_settings(conn, user_id, agent_id)
    task = build_task(settings, agent, target)

    counters = {"steps": 0, "leadsFound": 0}

    def on_step(idx, caption, url, png_bytes):
        path = _save_screenshot(job_id, idx, png_bytes)
        db.insert_job_step(conn, job_id, user_id, idx, caption, url, path)
        counters["steps"] = idx + 1
        db.update_job_result(conn, job_id, user_id, counters)

    prospects = asyncio.run(run_scrape(task, target, on_step))

    for p in prospects:
        d = p.model_dump()
        if db.lead_exists(conn, user_id, agent_id, d.get("profileUrl")):
            continue
        db.insert_lead(conn, user_id, agent_id, d)
        counters["leadsFound"] += 1
    db.finish_job(conn, job_id, user_id, counters)
    print(f"[worker] job {job_id} done — {counters['leadsFound']} leads, {counters['steps']} steps")


def main():
    conn = db.connect()
    print("[worker] scrape worker up — polling every", POLL_SECONDS, "s")
    while True:
        try:
            job = db.claim_next_scrape_job(conn)
            if not job:
                time.sleep(POLL_SECONDS)
                continue
            print("[worker] claimed job", job["id"])
            try:
                run_job(conn, job)
            except Exception as e:  # never let one bad job kill the loop
                print("[worker] job failed:", e)
                db.fail_job(conn, job["id"], job["user_id"], str(e))
        except KeyboardInterrupt:
            print("\n[worker] shutting down")
            break
        except Exception as e:
            print("[worker] loop error:", e)
            time.sleep(POLL_SECONDS)


if __name__ == "__main__":
    main()
```
